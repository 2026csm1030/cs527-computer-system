% --- Benchmark 1: Dynamic Array Summation ---
x1 = 0        % Index counter (i = 0)
x2 = 0        % Accumulated sum = 0
x3 = 0        % Memory pointer starting at 0x0

x4 = [x3]     % Read array size N from address 0

.loopback
x15 = x1 - x4 % Compare index x1 with array size x4 (updates NZCV flags)
BEQ .exit     % If index == N (Zero flag set), jump to .exit

x3 = x3 + 4   % Increment memory pointer by 4 bytes (next integer)
x6 = [x3]     % Load array element at address x3 into x6
x2 = x2 + x6  % Add element to total sum (x2)
x1 = x1 + 1   % Increment index counter (x1++)
BAL .loopback % Unconditional jump back to start of loop

.exit
x3 = x3 + 4   % Set target address to (N * 4 + 4)
[x3] = x2     % Store final calculated sum to memory
