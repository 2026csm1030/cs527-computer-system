Read x1, 0    // x1 = a
Read x2, 4    // x2 = b
Read x3, 8    // x3 = c
Read x4, 12   // x4 = d
x5 = x1 * x3  // x5 = ac
x6 = x2 * x4  // x6 = bd
x7 = x5 - x6  // x7 = Real Part (ac - bd)
Write x7, 16  // Store Real Part at address 16
x8 = x1 * x4  // x8 = ad
x9 = x2 * x3  // x9 = bc
x10 = x8 + x9 // x10 = Imaginary Part (ad + bc)
Write x10, 20 // Store Imaginary Part at address 20
