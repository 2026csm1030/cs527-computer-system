% --- Benchmark 2: FIR Filter (3-Tap) ---
x1 = 0         % Loop index i = 0
x3 = 0         % Input array pointer starting at address 0x0
x4 = [x3]      % Read input array length N from address 0

x10 = 16       % Output array pointer starting at address 0x10 (16)

.fir_loop
x15 = x1 - x4  % Compare current index x1 with N
BEQ .fir_end   % If i == N, exit loop

% Calculate address of X[i] at (0x4 + i*4)
x3 = x1 * 4
x3 = x3 + 4
x5 = [x3]      % Load current sample X[i] into x5

% Term 1: 1 * X[i]
x6 = x5 * 1

% Term 2: 2 * X[i-1] (Check if i >= 1)
x7 = 0
x16 = x1 - 1
BLT .skip_tap1
x20 = x3 - 4
x21 = [x20]    % Load X[i-1]
x7 = x21 * 2
.skip_tap1

% Term 3: 3 * X[i-2] (Check if i >= 2)
x8 = 0
x16 = x1 - 2
BLT .skip_tap2
x20 = x3 - 8
x22 = [x20]    % Load X[i-2]
x8 = x22 * 3
.skip_tap2

% Sum all three tap terms: Y[i] = Term1 + Term2 + Term3
x9 = x6 + x7
x9 = x9 + x8

% Store filtered result Y[i] to output array at address x10
[x10] = x9

x10 = x10 + 4  % Move output pointer to next word (0x10 + 4*i)
x1 = x1 + 1    % Increment loop index
BAL .fir_loop

.fir_end
